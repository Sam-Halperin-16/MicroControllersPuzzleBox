
    
  
/*
 * RP2040 capacitive screw touch test -- Raspberry Pi Pico C/C++ SDK.
 * Screw -> GP14; external 1 megaohm resistor from GP14 to GND.
 * LED -> GP15 through its existing current-limiting resistor.
 * Keep your finger off the screw until startup blinking finishes.
 * Link this source against pico_stdlib. No serial connection is needed.
 */
#include <stdbool.h>
#include <stdint.h>
#include "pico/stdlib.h"

#define TOUCH_PIN       14u
#define LED_PIN         15u
#define LED_ACTIVE_HIGH 1
#define SAMPLE_COUNT    16u
#define TIMEOUT_US      2000u
#define MIN_MARGIN_US   5u
#define MARGIN_PERCENT  30u
#define DEBOUNCE_COUNT  3u

static void set_led(bool on)
{
    gpio_put(LED_PIN, LED_ACTIVE_HIGH ? on : !on);
}

static uint32_t measure_touch(void)
{
    gpio_put(TOUCH_PIN, 1);
    gpio_set_dir(TOUCH_PIN, GPIO_OUT);
    sleep_us(10);

    uint32_t start = time_us_32();
    gpio_set_dir(TOUCH_PIN, GPIO_IN);
    while (gpio_get(TOUCH_PIN)) {
        if ((uint32_t)(time_us_32() - start) >= TIMEOUT_US) {
            break;
        }
    }
    uint32_t elapsed = time_us_32() - start;
    if (elapsed > TIMEOUT_US) elapsed = TIMEOUT_US;

    gpio_put(TOUCH_PIN, 0);
    gpio_set_dir(TOUCH_PIN, GPIO_OUT);
    sleep_us(100);
    return elapsed;
}

static uint32_t read_touch(void)
{
    uint32_t total = 0;
    for (uint32_t i = 0; i < SAMPLE_COUNT; ++i) {
        total += measure_touch();
    }
    return total / SAMPLE_COUNT;
}

int main(void)
{
    gpio_init(LED_PIN);
    gpio_set_dir(LED_PIN, GPIO_OUT);
    set_led(false);

    gpio_init(TOUCH_PIN);
    gpio_disable_pulls(TOUCH_PIN);
    gpio_put(TOUCH_PIN, 0);
    gpio_set_dir(TOUCH_PIN, GPIO_OUT);

    // Startup indication: leave the screw untouched.
    for (uint32_t i = 0; i < 5; ++i) {
        set_led(true);
        sleep_ms(100);
        set_led(false);
        sleep_ms(100);
    }

    // Keep blinking while measuring the untouched baseline.
    uint32_t total = 0;
    for (uint32_t i = 0; i < 64; ++i) {
        set_led((i / 8) % 2 == 0);
        total += read_touch();
        sleep_ms(10);
    }
    set_led(false);

    uint32_t baseline = total / 64;
    uint32_t margin = baseline * MARGIN_PERCENT / 100;
    if (margin < MIN_MARGIN_US) margin = MIN_MARGIN_US;
    const uint32_t on_threshold = baseline + margin;
    const uint32_t off_threshold = baseline + margin / 2;

    bool touched = false;
    uint32_t consecutive = 0;
    while (true) {
        uint32_t reading = read_touch();
        bool change = touched ? reading < off_threshold
                              : reading > on_threshold;
        if (change) {
            if (++consecutive >= DEBOUNCE_COUNT) {
                touched = !touched;
                consecutive = 0;
            }
        } else {
            consecutive = 0;
        }
        set_led(touched);
        sleep_ms(20);
    }
}
